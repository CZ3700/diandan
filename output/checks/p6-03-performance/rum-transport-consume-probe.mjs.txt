import { createServer } from 'node:http';
import { writeFile } from 'node:fs/promises';
import { setTimeout as delay } from 'node:timers/promises';
import { withNativeRumContext } from './apps/api/scripts/rum-browser-lifecycle.mjs';
const rows=[];
const server=createServer(async(req,res)=> { if(req.url==='/empty'){ for await(const _ of req){} await delay(80);res.writeHead(204);res.end();return;}res.setHeader('Content-Type','text/html');res.end('<!doctype html><title>Transport probe</title><h1>Transport probe</h1>'); });
await new Promise(r=>server.listen(0,'127.0.0.1',r));const origin=`http://127.0.0.1:${server.address().port}`;
try { await withNativeRumContext({},async(context)=> { const page=await context.newPage();let stage='setup'; const session=await context.newCDPSession(page);await session.send('Network.enable');const ids=new Set();session.on('Network.requestWillBeSent',e=>{if(e.request.url.endsWith('/empty')) {ids.add(e.requestId);rows.push({kind:'CDP-request',stage});}});session.on('Network.loadingFailed',e=>{if(ids.has(e.requestId))rows.push({kind:'CDP-failed',error:e.errorText,canceled:e.canceled,stage});});session.on('Network.loadingFinished',e=>{if(ids.has(e.requestId))rows.push({kind:'CDP-finished',stage});});
page.on('requestfailed',r=>{if(r.url().endsWith('/empty'))rows.push({kind:'PW-failed',error:r.failure()?.errorText,stage});});page.on('requestfinished',r=>{if(r.url().endsWith('/empty'))rows.push({kind:'PW-finished',stage});});page.on('response',r=>{if(r.url().endsWith('/empty'))rows.push({kind:'PW-response',status:r.status(),stage});});
await page.goto(origin);for(const keepalive of [false,true]){stage=`fetch-${keepalive}`;rows.push({kind:'fetch-resolution',keepalive,result:await page.evaluate(async(keepalive)=>{try{const r=await fetch('/empty',{method:'POST',keepalive,credentials:'omit',referrerPolicy:'no-referrer',cache:'no-store',body:'test'});return {status:r.status,body:await r.text()};}catch(e){return{error:e.name};}},keepalive)});await delay(500);}stage='void-fetch';await page.evaluate(()=>{void fetch('/empty',{method:'POST',keepalive:true,credentials:'omit',referrerPolicy:'no-referrer',cache:'no-store',headers:{'content-type':'application/json'},body:'{}'}).catch(()=>{});});await delay(1000);stage='await-status';rows.push({kind:'status-resolution',result:await page.evaluate(async()=>{const r=await fetch('/empty',{method:'POST',keepalive:true,body:'{}'});return r.status;})});await delay(500);
stage='void-status';await page.evaluate(()=>{void fetch('/empty',{method:'POST',keepalive:true,body:'{}'}).then(r=>{globalThis.transportStatus=r.status;}).catch(()=>{});});await delay(500);rows.push({kind:'void-status-resolution',result:await page.evaluate(()=>globalThis.transportStatus)});
stage='void-arrayBuffer';await page.evaluate(()=>{void fetch('/empty',{method:'POST',keepalive:true,body:'{}'}).then(r=>r.arrayBuffer()).catch(()=>{});});await delay(500);
stage='pre-cleanup';await delay(500);rows.push({kind:'before-cleanup'});}); }
finally {await new Promise(r=>server.close(r));await writeFile('output/checks/p6-03-performance/rum-transport-consume-probe.json',JSON.stringify(rows,null,2)+'\n');console.log(JSON.stringify(rows));}
