import { chromium } from '@playwright/test';
import { writeFile, mkdir } from 'node:fs/promises';
import { pathToFileURL } from 'node:url';
const html='/Users/mario/Desktop/.fan-support-regression/e2f45ec6-5775-483c-8eed-cbe2e40c1719/workspace/output/checks/p3-06-storefront-acceptance/run-2026-09-23T21-12-51-638Z/browser-attempt-1/rum/dashboard-cli/rum-dashboard.html';
const output='output/checks/p6-03-performance/rum-dashboard-locator-red';
await mkdir(output,{recursive:false});
const result={status:'RUNNING',pageErrors:[],cells:[]};
const browser=await chromium.launch({channel:'chrome',headless:true});
try {
 for(const viewport of [{width:390,height:844},{width:1440,height:900}]){
  const context=await browser.newContext({viewport});

   const cell={viewport,filterAccessibility:await page.locator('#filters').ariaSnapshot(),labels:await page.locator('select').evaluateAll(selects=>selects.map(select=>({labelText:[...select.labels].map(label=>label.textContent),selected:select.value}))),counts:{}};result.cells.push(cell);
   for(const field of ['locale','viewport'])cell.counts[field]={exactLabel:await page.getByLabel(field,{exact:true}).count(),partialLabel:await page.getByLabel(field).count(),exactRole:await page.getByRole('combobox',{name:field,exact:true}).count()};
   try{await page.getByLabel('locale',{exact:true}).selectOption('en',{timeout:1000});cell.originalSelector='UNEXPECTED_PASS';}catch(error){cell.originalSelector='REPRODUCED_TIMEOUT';cell.errorName=error.name;}
   await page.screenshot({path:`${output}/filters-${viewport.width}.png`,fullPage:true});
  }finally{await context.close();}
 }
 result.status=result.cells.every(cell=>cell.originalSelector==='REPRODUCED_TIMEOUT')?'EXPECTED_RED':'UNEXPECTED_RESULT';
}finally{await browser.close();await writeFile(`${output}/diagnostic.json`,JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result));}
