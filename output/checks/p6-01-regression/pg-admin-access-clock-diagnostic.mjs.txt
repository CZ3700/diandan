#!/usr/bin/env node
import assert from "node:assert/strict";
import { randomUUID, randomBytes } from "node:crypto";
import { Buffer } from "node:buffer";
import { setTimeout as delay } from "node:timers/promises";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { createRequire } from "node:module";
const { Client } = createRequire(new URL("../../../packages/persistence-postgres/package.json", import.meta.url))("pg");
import {
  createPostgresPersistence,
  runMigrations,
  withEphemeralPostgres,
} from "../../../packages/persistence-postgres/dist/index.js";
const workspaceRoot = path.resolve(
  path.dirname(fileURLToPath(import.meta.url)),
  "../../..",
);
const digest = () => randomBytes(32).toString("hex");
const issuer = "https://identity.example.test";
let checks = 0;
let stage = "schema";
let check = "none";
function equal(actual, expected, label) {
  check = label;
  assert.ok(JSON.stringify(actual) === JSON.stringify(expected), label);
  checks++;
}
function ok(value, label) {
  check = label;
  assert.ok(value, label);
  checks++;
}
await withEphemeralPostgres(async (config) => {
  await runMigrations({
    clientConfig: config,
    workspaceRoot,
    command: { direction: "up" },
  });
  const client = new Client(config);
  await client.connect();
  const persistence = createPostgresPersistence(config);
  const transaction = (work) =>
    persistence.adminAccessTransactionManager.runInAdminAccessTransaction(work);
  // Retrying an aborted serializable attempt is test orchestration, never provider exchange replay.
  async function run(method, command) {
    for (let attempt = 0; ; attempt++) {
      try {
        return await transaction(({ adminAccess }) =>
          adminAccess[method](command),
        );
      } catch (error) {
        if (
          error?.failure?.error?.code !== "TRANSACTION_ABORTED" ||
          attempt >= 3
        )
          throw error;
      }
    }
  }
  const createCommand = (ttlSeconds = 600) => ({
    schemaVersion: 1,
    requestId: randomUUID(),
    challengeId: randomUUID(),
    stateDigest: digest(),
    bindingDigest: digest(),
    configurationDigest: digest(),
    locale: "ja",
    ttlSeconds,
  });
  const claimCommand = (c) => ({
    schemaVersion: 1,
    requestId: randomUUID(),
    stateDigest: c.stateDigest,
    bindingDigest: c.bindingDigest,
    configurationDigest: c.configurationDigest,
    claimDigest: digest(),
  });
  const now = async () =>
    (
      await client.query(
        "SELECT to_char(clock_timestamp() AT TIME ZONE 'UTC','YYYY-MM-DD\"T\"HH24:MI:SS.US\"Z\"') AS now",
      )
    ).rows[0].now;
  async function challenge(ttl = 600) {
    const c = createCommand(ttl);
    equal((await run("create", c)).kind, "LOGIN_CREATED", "create challenge");
    const claim = claimCommand(c);
    equal((await run("claim", claim)).kind, "LOGIN_CLAIMED", "claim challenge");
    return { c, claim };
  }
  const identityId = randomUUID(),
    subjectDigest = digest();
  async function completion(flow, overrides = {}) {
    return {
      schemaVersion: 1,
      requestId: randomUUID(),
      challengeId: flow.c.challengeId,
      claimDigest: flow.claim.claimDigest,
      issuer,
      subjectDigest,
      authenticatedAt: await now(),
      mfa: true,
      sessionId: randomUUID(),
      sessionTokenDigest: digest(),
      csrfTokenDigest: digest(),
      sessionTtlSeconds: 600,
      maxAuthenticationAgeSeconds: 60,
      ...overrides,
    };
  }
  const logout = (session, revokeAll = false, overrides = {}) => ({
    schemaVersion: 1,
    requestId: randomUUID(),
    sessionTokenDigest: session.sessionTokenDigest,
    csrfTokenDigest: session.csrfTokenDigest,
    revokeAll,
    ...overrides,
  });
  async function rejectSql(
    sql,
    values = [],
    label = "invalid direct mutation rejected",
  ) {
    let failed = false;
    try {
      await client.query(sql, values);
    } catch (error) {
      failed = ["23514", "55000", "23503", "0A000"].includes(error.code);
    }
    ok(failed, label);
  }
  let blocker;
  try {
    equal(
      (
        await client.query(
          "SELECT to_regclass('public.admin_login_challenges') IS NOT NULL AS present",
        )
      ).rows[0].present,
      true,
      "durable login challenge schema is installed",
    );
    await client.query(
      "INSERT INTO admin_identities(id,issuer,external_subject_hash,status) VALUES($1,$2,$3,'ACTIVE')",
      [identityId, issuer, Buffer.from(subjectDigest, "hex")],
    );
    const expiredClaimed = await challenge(30),
      expiredReady = createCommand(30);
    await run("create", expiredReady);
    stage = "browser/config binding and single use";
    const c = createCommand();
    await run("create", c);
    const claim = claimCommand(c);
    for (const key of ["bindingDigest", "configurationDigest", "stateDigest"])
      equal(
        (await run("claim", { ...claim, [key]: digest() })).code,
        "LOGIN_RESTART_REQUIRED",
        "wrong binding/config/state cannot claim",
      );
    const claimed = await Promise.all([
      run("claim", claim),
      run("claim", { ...claim, claimDigest: digest() }),
    ]);
    equal(
      claimed.filter((r) => r.kind === "LOGIN_CLAIMED").length,
      1,
      "concurrent claim has one winner",
    );
    const winner = (
      await client.query(
        "SELECT encode(claim_digest,'hex') AS digest FROM admin_login_challenges WHERE id=$1",
        [c.challengeId],
      )
    ).rows[0].digest;
    const flow = { c, claim: { ...claim, claimDigest: winner } };
    const successful = await completion(flow);
    equal(
      (await run("complete", { ...successful, claimDigest: digest() })).code,
      "LOGIN_RESTART_REQUIRED",
      "wrong claim cannot mint session",
    );
    equal(
      (await run("complete", { ...successful, mfa: false })).code,
      "INVALID_COMMAND",
      "unverified MFA cannot enter persistence command",
    );
    const completed = await Promise.all([
      run("complete", successful),
      run("complete", successful),
    ]);
    equal(
      completed.filter((r) => r.kind === "SESSION_SAVED").length,
      1,
      "concurrent completion creates one session",
    );
    equal(
      (await run("complete", successful)).code,
      "LOGIN_RESTART_REQUIRED",
      "completed challenge cannot replay",
    );
    equal(
      (
        await client.query(
          "SELECT count(*)::int AS count FROM admin_sessions WHERE id=$1",
          [successful.sessionId],
        )
      ).rows[0].count,
      1,
      "exactly one session persisted",
    );
    equal(
      (
        await client.query(
          "SELECT count(*)::int AS count FROM admin_identity_roles WHERE admin_identity_id=$1",
          [identityId],
        )
      ).rows[0].count,
      0,
      "login creates no grants",
    );
    const read =
      await persistence.adminSessionTransactionManager.runInAdminSessionTransaction(
        ({ adminSession }) =>
          adminSession.read({
            schemaVersion: 1,
            sessionTokenDigest: successful.sessionTokenDigest,
            csrfTokenDigest: successful.csrfTokenDigest,
          }),
      );
    equal(read.outcome, "SUCCESS", "existing reader accepts fresh session");
    stage = "denials consume challenges and audit without subject claims";
    for (const overrides of [
      { subjectDigest: digest() },
      { issuer: "https://other.example.test" },
    ]) {
      const f = await challenge();
      const cmd = await completion(f, overrides);
      equal(
        (await run("complete", cmd)).code,
        "ACCESS_DENIED",
        "unknown identity is denied",
      );
      equal(
        (await run("complete", cmd)).code,
        "LOGIN_RESTART_REQUIRED",
        "denied challenge stays consumed",
      );
    }
    await client.query(
      "UPDATE admin_identities SET status='SUSPENDED' WHERE id=$1",
      [identityId],
    );
    const suspended = await completion(await challenge());
    equal(
      (await run("complete", suspended)).code,
      "ACCESS_DENIED",
      "suspended identity denied",
    );
    await client.query(
      "UPDATE admin_identities SET status='ACTIVE' WHERE id=$1",
      [identityId],
    );
    stage = "logout-all lock-time expiry";
    const blockedSession = {
      sessionId: randomUUID(),
      sessionTokenDigest: digest(),
      csrfTokenDigest: digest(),
    };
    await client.query(
      `INSERT INTO admin_sessions(id,admin_identity_id,session_token_digest,csrf_token_digest,authenticated_with_mfa,created_at,expires_at)
      VALUES($1,$2,$3,$4,true,clock_timestamp(),clock_timestamp()+interval '2 seconds')`,
      [
        blockedSession.sessionId,
        identityId,
        Buffer.from(blockedSession.sessionTokenDigest, "hex"),
        Buffer.from(blockedSession.csrfTokenDigest, "hex"),
      ],
    );
    blocker = new Client(config);
    await blocker.connect();
    await blocker.query("BEGIN");
    await blocker.query("SELECT id FROM admin_sessions WHERE id=$1 FOR SHARE", [
      blockedSession.sessionId,
    ]);
    const revokeBlockerPid = (
      await blocker.query("SELECT pg_backend_pid() AS pid")
    ).rows[0].pid;
    const blockedRevoke = transaction(({ adminAccess }) =>
      adminAccess.revoke(logout(blockedSession, true)),
    ).catch(() => ({ code: "HARNESS_FAILURE" }));
    let revokeWaiting = false;
    for (let attempt = 0; attempt < 100; attempt++) {
      revokeWaiting = (
        await client.query(
          "SELECT EXISTS(SELECT 1 FROM pg_stat_activity WHERE $1=ANY(pg_blocking_pids(pid))) AS waiting",
          [revokeBlockerPid],
        )
      ).rows[0].waiting;
      if (revokeWaiting) break;
      await delay(10);
    }
    ok(revokeWaiting, "logout-all is observably waiting on session row lock");
    const revokeWait = (
      await client.query(
        "SELECT greatest(0,extract(epoch from expires_at-clock_timestamp())+0.03) AS seconds FROM admin_sessions WHERE id=$1",
        [blockedSession.sessionId],
      )
    ).rows[0].seconds;
    await delay(Number(revokeWait) * 1000);
    await blocker.query("COMMIT");
    await blocker.end();
    blocker = undefined;
    equal(
      (await blockedRevoke).code,
      "UNAUTHENTICATED",
      "logout-all waiting for row lock rechecks database expiry",
    );
    equal(
      (
        await client.query(
          "SELECT revoked_at IS NULL AS unchanged FROM admin_sessions WHERE id=$1",
          [blockedSession.sessionId],
        )
      ).rows[0].unchanged,
      true,
      "expired blocked logout-all does not mutate session",
    );
    for (const authenticatedAt of [
      "2000-01-01T00:00:00.000Z",
      "2099-01-01T00:00:00.000Z",
    ]) {
      const cmd = await completion(await challenge(), { authenticatedAt });
      equal(
        (await run("complete", cmd)).code,
        "ACCESS_DENIED",
        "stale/future authentication denied",
      );
      equal(
        (await run("complete", cmd)).code,
        "LOGIN_RESTART_REQUIRED",
        "freshness denial consumed",
      );
    }
    for (const reasonCode of [
      "IDENTITY_REJECTED",
      "MFA_REQUIRED",
      "AUTHENTICATION_EXPIRED",
    ]) {
      const f = await challenge();
      const cmd = {
        schemaVersion: 1,
        requestId: randomUUID(),
        challengeId: f.c.challengeId,
        claimDigest: f.claim.claimDigest,
        reasonCode,
      };
      equal(
        (await run("reject", cmd)).kind,
        "LOGIN_REJECTED",
        "adapter rejection is durably finalized",
      );
      equal(
        (await run("complete", await completion(f))).code,
        "LOGIN_RESTART_REQUIRED",
        "rejected provider result cannot replay",
      );
    }
    stage = "CSRF and revocation";
    equal(
      (
        await run(
          "revoke",
          logout(successful, false, { csrfTokenDigest: digest() }),
        )
      ).code,
      "CSRF_INVALID",
      "tampered CSRF cannot revoke",
    );
    equal(
      (
        await run(
          "revoke",
          logout(successful, false, { sessionTokenDigest: digest() }),
        )
      ).code,
      "UNAUTHENTICATED",
      "unknown session is rejected",
    );
    const second = await completion(await challenge());
    await run("complete", second);
    equal(
      (await run("revoke", logout(successful))).kind,
      "LOGGED_OUT",
      "current logout succeeds",
    );
    const auditsAfterLogout = (
      await client.query("SELECT count(*)::int AS n FROM audit_logs")
    ).rows[0].n;
    equal(
      (await run("revoke", logout(successful))).kind,
      "LOGGED_OUT",
      "current logout repeats idempotently",
    );
    equal(
      (await client.query("SELECT count(*)::int AS n FROM audit_logs")).rows[0]
        .n,
      auditsAfterLogout,
      "repeat current logout adds no duplicate audit",
    );
    equal(
      (
        await client.query(
          "SELECT revoked_at IS NULL AS live FROM admin_sessions WHERE id=$1",
          [second.sessionId],
        )
      ).rows[0].live,
      true,
      "current logout preserves other session",
    );
    equal(
      (await run("revoke", logout(successful, true))).code,
      "UNAUTHENTICATED",
      "revoked credential cannot revoke later sessions",
    );
    const third = await completion(await challenge());
    await run("complete", third);
    const revocations = await Promise.all([
      run("revoke", logout(second, true)),
      run("revoke", logout(third)),
    ]);
    ok(
      revocations.every(
        (r) => r.kind === "LOGGED_OUT" || r.code === "UNAUTHENTICATED",
      ),
      "concurrent logout is fail closed",
    );
    equal(
      (
        await client.query(
          "SELECT count(*)::int AS n FROM admin_sessions WHERE admin_identity_id=$1 AND revoked_at IS NULL",
          [identityId],
        )
      ).rows[0].n,
      0,
      "all existing sessions are revoked",
    );
    stage = "expired current-session logout";
    const expiredSession = {
      sessionId: randomUUID(),
      sessionTokenDigest: digest(),
      csrfTokenDigest: digest(),
    };
    await client.query(
      `INSERT INTO admin_sessions(id,admin_identity_id,session_token_digest,csrf_token_digest,authenticated_with_mfa,created_at,expires_at)
      VALUES($1,$2,$3,$4,true,clock_timestamp()-interval '120 seconds',clock_timestamp()-interval '60 seconds')`,
      [
        expiredSession.sessionId,
        identityId,
        Buffer.from(expiredSession.sessionTokenDigest, "hex"),
        Buffer.from(expiredSession.csrfTokenDigest, "hex"),
      ],
    );
    equal(
      (await run("revoke", logout(expiredSession, true))).code,
      "UNAUTHENTICATED",
      "expired session cannot revoke all sessions",
    );
    equal(
      (
        await run(
          "revoke",
          logout(expiredSession, false, { csrfTokenDigest: digest() }),
        )
      ).code,
      "CSRF_INVALID",
      "expired current logout still requires CSRF",
    );
    await client.query(
      "UPDATE admin_identities SET status='SUSPENDED' WHERE id=$1",
      [identityId],
    );
    equal(
      (await run("revoke", logout(expiredSession))).kind,
      "LOGGED_OUT",
      "expired suspended identity can revoke own session",
    );
    const expiredAuditCount = (
      await client.query(
        "SELECT count(*)::int AS n FROM audit_logs WHERE subject_id=$1",
        [expiredSession.sessionId],
      )
    ).rows[0].n;
    equal(
      (await run("revoke", logout(expiredSession))).kind,
      "LOGGED_OUT",
      "expired current logout repeats idempotently",
    );
    equal(
      (
        await client.query(
          "SELECT count(*)::int AS n FROM audit_logs WHERE subject_id=$1",
          [expiredSession.sessionId],
        )
      ).rows[0].n,
      expiredAuditCount,
      "repeat expired current logout adds no audit",
    );
    equal(
      (
        await client.query(
          "SELECT revoked_at IS NOT NULL AS revoked FROM admin_sessions WHERE id=$1",
          [expiredSession.sessionId],
        )
      ).rows[0].revoked,
      true,
      "expired session actually revoked",
    );
    await client.query(
      "UPDATE admin_identities SET status='ACTIVE' WHERE id=$1",
      [identityId],
    );
    stage = "atomicity and immutable evidence";
    const rollbackFlow = await challenge(),
      rollbackCommand = await completion(rollbackFlow);
    await assert.rejects(
      transaction(async ({ adminAccess }) => {
        await adminAccess.complete(rollbackCommand);
        throw new Error("intentional rollback");
      }),
    );
    checks++;
    equal(
      (
        await client.query(
          "SELECT count(*)::int AS n FROM admin_sessions WHERE id=$1",
          [rollbackCommand.sessionId],
        )
      ).rows[0].n,
      0,
      "failed transaction saves no session",
    );
    equal(
      (await run("complete", rollbackCommand)).kind,
      "SESSION_SAVED",
      "rolled back claim consumption is atomic",
    );
    stage = "linked session evidence guards";
    const otherIdentityId = randomUUID();
    await client.query(
      "INSERT INTO admin_identities(id,issuer,external_subject_hash,status) VALUES($1,$2,$3,'ACTIVE')",
      [otherIdentityId, issuer, Buffer.from(digest(), "hex")],
    );
    for (const [field, expression, value] of [
      ["session_token_digest", "$2", Buffer.from(digest(), "hex")],
      ["csrf_token_digest", "$2", Buffer.from(digest(), "hex")],
      ["admin_identity_id", "$2", otherIdentityId],
      ["authenticated_with_mfa", "false", null],
      ["created_at", "created_at-interval '1 second'", null],
      ["expires_at", "expires_at+interval '1 second'", null],
      ["last_seen_at", "last_seen_at-interval '1 second'", null],
    ]) {
      await rejectSql(
        `UPDATE admin_sessions SET ${field}=${expression} WHERE id=$1`,
        value === null
          ? [rollbackCommand.sessionId]
          : [rollbackCommand.sessionId, value],
        `linked session ${field} cannot be rewritten`,
      );
    }
    await rejectSql(
      "UPDATE admin_sessions SET revoked_at=clock_timestamp() WHERE id=$1",
      [rollbackCommand.sessionId],
      "linked session revocation requires exact audit",
    );
    await rejectSql(
      "UPDATE admin_sessions SET revoked_at=NULL WHERE id=$1",
      [successful.sessionId],
      "linked session cannot be resurrected",
    );
    await rejectSql(
      "DELETE FROM admin_sessions WHERE id=$1",
      [rollbackCommand.sessionId],
      "linked session cannot be deleted",
    );
    await rejectSql(
      "TRUNCATE admin_sessions CASCADE",
      [],
      "linked session cannot be truncated",
    );
    for (const mismatch of [
      "actor",
      "subject",
      "time",
      "action",
      "correlation",
      "request",
      "reason",
    ]) {
      let failed = false;
      await client.query("BEGIN");
      try {
        const revokedAt = await now();
        await client.query(
          `INSERT INTO audit_logs(id,actor_type,actor_id,action,subject_type,subject_id,reason_code,request_id,correlation_id,outcome,created_at)
          VALUES($1,'ADMIN',$2,$3,'ADMIN_SESSION',$4,$5,$6,$7,'SUCCEEDED',$8::timestamptz)`,
          [
            randomUUID(),
            mismatch === "actor" ? otherIdentityId : identityId,
            mismatch === "action"
              ? "UNRELATED_ACTION"
              : "ADMIN_SESSION_REVOKED",
            mismatch === "subject" ? randomUUID() : rollbackCommand.sessionId,
            mismatch === "reason" ? "UNRELATED_REASON" : "USER_LOGOUT",
            mismatch === "request" ? null : randomUUID(),
            mismatch === "correlation"
              ? randomUUID()
              : rollbackCommand.sessionId,
            mismatch === "time" ? "2000-01-01T00:00:00Z" : revokedAt,
          ],
        );
        await client.query(
          "UPDATE admin_sessions SET revoked_at=$2::timestamptz WHERE id=$1",
          [rollbackCommand.sessionId, revokedAt],
        );
        await client.query("COMMIT");
      } catch (error) {
        failed = ["23514", "55000"].includes(error.code);
      } finally {
        await client.query("ROLLBACK");
      }
      ok(failed, `linked revocation rejects mismatched audit ${mismatch}`);
    }
    const boundaryRejections = [];
    for (const boundary of ["current-correlation", "future-revocation"]) {
      let rejected = false;
      await client.query("BEGIN");
      try {
        const eventTime =
          boundary === "future-revocation"
            ? (
                await client.query(
                  `SELECT to_char((clock_timestamp()+interval '1 day') AT TIME ZONE 'UTC','YYYY-MM-DD"T"HH24:MI:SS.US"Z"') AS time`,
                )
              ).rows[0].time
            : await now();
        const anchorId =
          boundary === "current-correlation"
            ? randomUUID()
            : rollbackCommand.sessionId;
        if (boundary === "current-correlation") {
          // A distinct same-owner, same-event anchor passes the earlier broad anchor rule.
          await client.query(
            `INSERT INTO admin_sessions(id,admin_identity_id,session_token_digest,csrf_token_digest,authenticated_with_mfa,created_at,expires_at,revoked_at)
            VALUES($1,$2,$3,$4,true,$5::timestamptz-interval '1 second',$5::timestamptz+interval '600 seconds',$5::timestamptz)`,
            [
              anchorId,
              identityId,
              Buffer.from(digest(), "hex"),
              Buffer.from(digest(), "hex"),
              eventTime,
            ],
          );
        }
        await client.query(
          `INSERT INTO audit_logs(id,actor_type,actor_id,action,subject_type,subject_id,reason_code,request_id,correlation_id,outcome,created_at)
          VALUES($1,'ADMIN',$2,'ADMIN_SESSION_REVOKED','ADMIN_SESSION',$3,'USER_LOGOUT',$4,$5,'SUCCEEDED',$6::timestamptz)`,
          [
            randomUUID(),
            identityId,
            rollbackCommand.sessionId,
            randomUUID(),
            anchorId,
            eventTime,
          ],
        );
        await client.query(
          "UPDATE admin_sessions SET revoked_at=$2::timestamptz WHERE id=$1",
          [rollbackCommand.sessionId, eventTime],
        );
        await client.query("SET CONSTRAINTS ALL IMMEDIATE");
      } catch (error) {
        rejected = ["23514", "55000"].includes(error.code);
      } finally {
        await client.query("ROLLBACK");
      }
      boundaryRejections.push(rejected);
    }
    equal(
      boundaryRejections,
      [true, true],
      `current self-correlation / future revocation rejected: ${boundaryRejections.join("/")}`,
    );
    await client.query("DELETE FROM admin_identities WHERE id=$1", [
      otherIdentityId,
    ]);
    await client.query(
      "UPDATE admin_sessions SET last_seen_at=clock_timestamp() WHERE id=$1",
      [rollbackCommand.sessionId],
    );
    checks++;
    const allIssued = [rollbackCommand];
    for (let index = 0; index < 2; index++) {
      const issued = await completion(await challenge());
      equal(
        (await run("complete", issued)).kind,
        "SESSION_SAVED",
        "additional linked session saved",
      );
      allIssued.push(issued);
    }
    const allCommand = logout(allIssued[0], true);
    equal(
      (await run("revoke", allCommand)).kind,
      "LOGGED_OUT",
      "all linked sessions revoked with individual audit",
    );
    const issuedAudits = (
      await client.query(
        `SELECT s.id,count(a.id)::int AS audits FROM admin_sessions s
      LEFT JOIN audit_logs a ON a.subject_id=s.id AND a.actor_id=s.admin_identity_id
      AND a.action='ADMIN_SESSIONS_REVOKED' AND a.created_at=s.revoked_at AND a.request_id=$2
      AND a.correlation_id=$3 GROUP BY s.id HAVING s.id=ANY($1::uuid[])`,
        [
          allIssued.map((s) => s.sessionId),
          allCommand.requestId,
          allIssued[0].sessionId,
        ],
      )
    ).rows;
    equal(
      issuedAudits.length,
      allIssued.length,
      "all target session evidence is present",
    );
    ok(
      issuedAudits.every((row) => row.audits === 1),
      "every revoked linked session has one matching immutable audit",
    );
    await client.query(
      "UPDATE admin_sessions SET csrf_token_digest=$2 WHERE id=$1",
      [expiredSession.sessionId, Buffer.from(digest(), "hex")],
    );
    checks++;
    await rejectSql(
      "UPDATE admin_login_challenges SET state='READY' WHERE id=$1",
      [flow.c.challengeId],
    );
    await rejectSql(
      "UPDATE admin_login_challenges SET binding_digest=$2 WHERE id=$1",
      [flow.c.challengeId, Buffer.from(digest(), "hex")],
    );
    await rejectSql("DELETE FROM admin_login_challenges WHERE id=$1", [
      flow.c.challengeId,
    ]);
    await rejectSql(
      "UPDATE audit_logs SET outcome='FAILED' WHERE action='ADMIN_LOGIN_SUCCEEDED'",
    );
    await rejectSql(
      "DELETE FROM audit_logs WHERE action='ADMIN_LOGIN_REJECTED'",
    );
    await rejectSql("TRUNCATE audit_logs");
    stage = "database-clock expiry";
    blocker = new Client(config);
    await blocker.connect();
    await blocker.query("BEGIN");
    await blocker.query(
      "SELECT id FROM admin_login_challenges WHERE id=$1 FOR UPDATE",
      [expiredReady.challengeId],
    );
    const blockedClaim = transaction(({ adminAccess }) =>
      adminAccess.claim(claimCommand(expiredReady)),
    ).catch((error) => {
      process.stdout.write(JSON.stringify({ diagnostic: "BLOCKED_CLAIM_ERROR", name: error?.name, code: error?.failure?.error?.code ?? null }) + "\n");
      return { code: "HARNESS_FAILURE" };
    });
    const blockerPid = (await blocker.query("SELECT pg_backend_pid() AS pid"))
      .rows[0].pid;
    let waiting = false;
    for (let attempt = 0; attempt < 100; attempt++) {
      waiting = (
        await client.query(
          "SELECT EXISTS(SELECT 1 FROM pg_stat_activity WHERE $1=ANY(pg_blocking_pids(pid))) AS waiting",
          [blockerPid],
        )
      ).rows[0].waiting;
      if (waiting) break;
      await delay(20);
    }
    ok(waiting, "claim is observably waiting on the held challenge row");
    const wait = (
      await client.query(
        "SELECT greatest(0,extract(epoch from max(expires_at)-clock_timestamp())+0.03) AS seconds FROM admin_login_challenges WHERE id=ANY($1::uuid[])",
        [[expiredClaimed.c.challengeId, expiredReady.challengeId]],
      )
    ).rows[0].seconds;
    process.stdout.write(JSON.stringify({ diagnostic: "WAIT_REQUEST", seconds: Number(wait) }) + "\n");
    const waitStart = performance.now();
    await delay(Number(wait) * 1000);
    const deadlineBeforeUnlock = (await client.query("SELECT extract(epoch FROM max(expires_at) - clock_timestamp()) AS remaining FROM admin_login_challenges WHERE id=ANY($1::uuid[])", [[expiredClaimed.c.challengeId, expiredReady.challengeId]])).rows[0];
    process.stdout.write(JSON.stringify({ diagnostic: "BEFORE_UNLOCK", hostElapsedMs: performance.now() - waitStart, databaseRemainingSeconds: Number(deadlineBeforeUnlock.remaining) }) + "\n");
    await blocker.query("COMMIT");
    await blocker.end();
    blocker = undefined;
    const observedBlockedClaim = await blockedClaim;
    const clockDiagnostic = (await client.query("SELECT state, extract(epoch FROM claimed_at-expires_at) AS claimed_after_expiry_seconds, extract(epoch FROM clock_timestamp()-expires_at) AS now_after_expiry_seconds FROM admin_login_challenges WHERE id=$1", [expiredReady.challengeId])).rows[0];
    process.stdout.write(JSON.stringify({ diagnostic: "BLOCKED_CLAIM_RESULT", code: observedBlockedClaim.code ?? null, kind: observedBlockedClaim.kind ?? null, ...clockDiagnostic }) + "\n");
    equal(
      observedBlockedClaim.code,
      "LOGIN_RESTART_REQUIRED",
      "claim waiting for lock rechecks database expiry",
    );
    equal(
      (await run("claim", claimCommand(expiredReady))).code,
      "LOGIN_RESTART_REQUIRED",
      "expired READY challenge cannot claim",
    );
    equal(
      (await run("complete", await completion(expiredClaimed))).code,
      "LOGIN_RESTART_REQUIRED",
      "expired CLAIMED challenge cannot mint session",
    );
    equal(
      (
        await client.query(
          "SELECT state FROM admin_login_challenges WHERE id=$1",
          [expiredClaimed.c.challengeId],
        )
      ).rows[0].state,
      "CONSUMED",
      "expired completion is finalized",
    );
    equal(
      (await client.query("SELECT count(*)::int AS n FROM admin_identities"))
        .rows[0].n,
      1,
      "login never creates identity",
    );
    let downgradeDenied = false;
    try {
      await runMigrations({
        clientConfig: config,
        workspaceRoot,
        command: { direction: "down", confirmVersion: "0030" },
      });
    } catch {
      downgradeDenied = true;
    }
    ok(downgradeDenied, "durable access history prevents downgrade");
    process.stdout.write(JSON.stringify({ result: "PASS", checks }) + "\n");
  } catch (error) {
    process.stderr.write(
      JSON.stringify({
        result: "FAIL",
        stage,
        checks,
        check,
        failure:
          error?.name === "AssertionError"
            ? "ASSERTION"
            : "PERSISTENCE_OR_HARNESS",
      }) + "\n",
    );
    process.exitCode = 1;
  } finally {
    if (blocker) {
      await blocker.query("ROLLBACK");
      await blocker.end();
    }
    await persistence.close();
    await client.end();
  }
});
